每个实验需要约500k步训练，预计每个实验需要数小时到数十小时（取决于硬件）。

```bash
cd /opt/03-ai/01-proj/Neurosurgeon/rl_collaborative_inference
source /opt/03-ai/01-proj/Neurosurgeon/neurosurgeon_env/bin/activate

# 为每个seed补充缺失的实验（后台运行）
for SEED in 2 3; do
  EXP_DIR="experiments_paper/seed_${SEED}/ablation_20251127_055835"
  
  # 补充 ablation 实验（后台运行）
  nohup python run_missing_experiments.py \
    --output_dir "${EXP_DIR}" \
    --seed ${SEED} \
    --data_dir ../data/caltech-101 \
    --run_ablation \
    --max_steps_rl 500000 \
    > logs/missing_ablation_seed_${SEED}.log 2>&1 &
  
  # 补充 multi-user 实验（后台运行）
  nohup python run_missing_experiments.py \
    --output_dir "${EXP_DIR}" \
    --seed ${SEED} \
    --data_dir ../data/caltech-101 \
    --run_multi_user \
    --max_steps_multi_user 500000 \
    > logs/missing_multi_user_seed_${SEED}.log 2>&1 &
done

# 检查运行状态
ps aux | grep run_missing_experiments
tail -f logs/missing_ablation_seed_2.log
```
