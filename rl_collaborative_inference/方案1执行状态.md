### 步骤2：快速测试模型准确率

```bash
source ../neurosurgeon_env/bin/activate
python -c "
import torch
from models.AlexNet import AlexNet
from src.dataset_loader import get_caltech101_dataloader

model = AlexNet(3, 101)
model.load_state_dict(torch.load('./alexnet_caltech101.pth'))
model.eval()

_, test = get_caltech101_dataloader('../data/caltech-101', batch_size=32, split='test')
correct = 0
total = 0
for images, labels in test:
    with torch.no_grad():
        outputs = model(images)
        _, pred = torch.max(outputs, 1)
        correct += (pred == labels).sum().item()
        total += labels.size(0)
    if total >= 200:
        break

print(f'Test Accuracy: {100*correct/total:.2f}%')
"
```
